# 配置管理设计文档

## 1. 概述

配置管理负责存储和管理系统运行所需的各类配置，包括：
- 模型配置（大模型 API）
- MCP 配置（知识库服务）
- Agent 配置（工作流预设）
- 系统配置（全局参数）

## 2. 配置存储结构

```
agent-runner/
├── config/
│   ├── model-config.json      # 模型配置
│   ├── mcp-config.json        # MCP 配置
│   ├── agent-presets.json     # Agent 预设
│   └── system-config.json     # 系统配置
```

## 4. 配置管理器

### 4.1 核心实现

```javascript
class ConfigManager {
    constructor() {
        this.configDir = path.join(__dirname, '../config');
        this.encryptionKey = process.env.AGENT_RUNNER_KEY || 'default-key';
        this.cache = new Map();
    }
    
    /**
     * 加载配置
     */
    async load(configName) {
        // 检查缓存
        if (this.cache.has(configName)) {
            return this.cache.get(configName);
        }
        
        const filePath = path.join(this.configDir, `${configName}.json`);
        
        if (!fs.existsSync(filePath)) {
            return null;
        }
        
        const content = fs.readFileSync(filePath, 'utf-8');
        const config = JSON.parse(content);
        
        // 解密敏感字段
        this.decryptConfig(config);
        
        this.cache.set(configName, config);
        return config;
    }
    
    /**
     * 保存配置
     */
    async save(configName, config) {
        const filePath = path.join(this.configDir, `${configName}.json`);
        
        // 确保目录存在
        if (!fs.existsSync(this.configDir)) {
            fs.mkdirSync(this.configDir, { recursive: true });
        }
        
        // 加密敏感字段
        const configToSave = this.encryptConfig({ ...config });
        
        fs.writeFileSync(filePath, JSON.stringify(configToSave, null, 2));
        
        // 更新缓存
        this.cache.set(configName, config);
    }
    
    /**
     * 加密配置
     */
    encryptConfig(config) {
        const encrypted = { ...config };
        
        if (encrypted.api_key) {
            encrypted.api_key_encrypted = this.encrypt(encrypted.api_key);
            delete encrypted.api_key;
        }
        
        return encrypted;
    }
    
    /**
     * 解密配置
     */
    decryptConfig(config) {
        if (config.api_key_encrypted) {
            config.api_key = this.decrypt(config.api_key_encrypted);
            delete config.api_key_encrypted;
        }
    }
    
    /**
     * AES-256 加密
     */
    encrypt(text) {
        const cipher = crypto.createCipher('aes-256-cbc', this.encryptionKey);
        let encrypted = cipher.update(text, 'utf8', 'hex');
        encrypted += cipher.final('hex');
        return encrypted;
    }
    
    /**
     * AES-256 解密
     */
    decrypt(encrypted) {
        const decipher = crypto.createDecipher('aes-256-cbc', this.encryptionKey);
        let decrypted = decipher.update(encrypted, 'hex', 'utf8');
        decrypted += decipher.final('utf8');
        return decrypted;
    }
    
    /**
     * 获取模型配置
     */
    async getModelConfig() {
        return this.load('model-config');
    }
    
    /**
     * 获取 MCP 配置
     */
    async getMCPConfig() {
        return this.load('mcp-config');
    }
    
    /**
     * 获取 Agent 预设
     */
    async getAgentPreset(presetId) {
        const presets = await this.load('agent-presets');
        return presets.presets.find(p => p.id === presetId);
    }
    
    /**
     * 保存 Agent 预设
     */
    async saveAgentPreset(preset) {
        const presets = await this.load('agent-presets') || { presets: [] };
        
        const index = presets.presets.findIndex(p => p.id === preset.id);
        if (index >= 0) {
            presets.presets[index] = preset;
        } else {
            presets.presets.push(preset);
        }
        
        await this.save('agent-presets', presets);
    }
}
```

## 5. 配置验证

```javascript
class ConfigValidator {
    /**
     * 验证模型配置
     */
    static validateModelConfig(config) {
        const errors = [];
        
        if (!config.provider) {
            errors.push('provider is required');
        }
        
        if (!config.api_key && !config.api_key_encrypted) {
            errors.push('api_key is required');
        }
        
        if (!config.model) {
            errors.push('model is required');
        }
        
        if (config.temperature !== undefined) {
            if (config.temperature < 0 || config.temperature > 2) {
                errors.push('temperature must be between 0 and 2');
            }
        }
        
        return {
            valid: errors.length === 0,
            errors
        };
    }
    
    /**
     * 验证 MCP 配置
     */
    static validateMCPConfig(config) {
        const errors = [];
        
        if (!config.server_url) {
            errors.push('server_url is required');
        }
        
        if (!config.api_key && !config.api_key_encrypted) {
            errors.push('api_key is required');
        }
        
        return {
            valid: errors.length === 0,
            errors
        };
    }
    
    /**
     * 验证 Agent 预设
     */
    static validateAgentPreset(preset) {
        const errors = [];
        
        if (!preset.id) {
            errors.push('id is required');
        }
        
        if (!preset.name) {
            errors.push('name is required');
        }
        
        if (!preset.steps || preset.steps.length === 0) {
            errors.push('at least one step is required');
        }
        
        return {
            valid: errors.length === 0,
            errors
        };
    }
}
```

## 6. 配置初始化

```javascript
/**
 * 初始化默认配置
 */
async function initializeDefaultConfig() {
    const configManager = new ConfigManager();
    
    // 初始化模型配置
    const modelConfig = await configManager.getModelConfig();
    if (!modelConfig) {
        await configManager.save('model-config', {
            provider: 'openai',
            model: 'gpt-4',
            base_url: 'https://api.openai.com/v1',
            temperature: 0.7,
            max_tokens: 4000
        });
    }
    
    // 初始化 MCP 配置
    const mcpConfig = await configManager.getMCPConfig();
    if (!mcpConfig) {
        await configManager.save('mcp-config', {
            server_url: 'http://localhost:8080',
            timeout: 30000,
            cache: {
                enabled: true,
                ttl: 3600
            }
        });
    }
    
    // 初始化 Agent 预设
    const presets = await configManager.load('agent-presets');
    if (!presets) {
        await configManager.save('agent-presets', {
            presets: [
                {
                    id: 'default',
                    name: '默认4步流程',
                    steps: [
                        { name: 'planner', agent: 'planner', config: {} },
                        { name: 'retriever', agent: 'retriever', config: {} },
                        { name: 'generator', agent: 'generator', config: {} },
                        { name: 'validator', agent: 'validator', config: {} }
                    ]
                }
            ],
            default_preset: 'default'
        });
    }
}
```

## 8. 配置热更新

配置修改后自动生效，无需重启服务：

```javascript
// 监听配置文件变化
fs.watch(configDir, (eventType, filename) => {
    if (filename.endsWith('.json')) {
        const configName = filename.replace('.json', '');
        configManager.cache.delete(configName);
        console.log(`Config ${configName} reloaded`);
    }
});
```

## 9. MCP Headers 配置（新增）

### 9.1 概述

MCP 服务器通常需要自定义 HTTP 标头进行认证和传递元数据。系统支持配置多个自定义 headers，敏感标头可加密存储。

### 9.2 配置结构

```json
{
  "server_url": "http://localhost:8080",
  "api_key_encrypted": "U2FsdGVkU1...",
  "timeout": 30000,
  "retry_count": 3,
  "cache": {
    "enabled": true,
    "ttl": 3600
  },
  "headers": [
    {
      "name": "Authorization",
      "value_encrypted": "U2FsdGVkU1...",
      "encrypted": true,
      "description": "Bearer token 认证"
    },
    {
      "name": "X-Request-ID",
      "value": "req-12345",
      "encrypted": false,
      "description": "请求追踪 ID"
    }
  ]
}
```

### 9.3 字段说明

| 字段 | 类型 | 必填 | 说明 |
|------|------|------|------|
| name | string | 是 | HTTP 标头名称 |
| value | string | 否 | 标头值（明文，encrypted=false 时使用） |
| value_encrypted | string | 否 | 加密后的标头值（encrypted=true 时使用） |
| encrypted | boolean | 是 | 是否加密存储 |
| description | string | 否 | 标头描述（用于 UI 显示） |

### 9.4 加密逻辑

```javascript
// 加密敏感 headers
if (header.encrypted && header.value) {
  header.value_encrypted = this._encrypt(header.value);
  delete header.value;
}

// 解密敏感 headers
if (header.encrypted && header.value_encrypted) {
  header.value = this._decrypt(header.value_encrypted);
}
```
