# 新功能模块设计文档

## 1. 概述

本文档详细描述四个新增功能的模块设计：
1. **Markdown 预览**：提示词内容的实时渲染预览
2. **大纲上传**：支持上传知识点大纲文档
3. **执行方案选择**：配置和选择不同的 Agent 执行流程
4. **文档查看**：在前端查看生成的文档

## 2. 功能一：Markdown 预览

### 2.1 需求描述

- 生成的提示词内容为 Markdown 格式
- 前端需要渲染为 HTML 展示
- 内容需要完全可见（支持滚动）
- 支持代码高亮

### 2.2 技术方案

**依赖库：**
- `marked.js`：Markdown 解析器
- `highlight.js`：代码语法高亮

**CDN 引入：**
```html
<!-- Markdown 渲染 -->
<script src="https://cdn.jsdelivr.net/npm/marked/marked.min.js"></script>
<!-- 代码高亮 -->
<link rel="stylesheet" href="https://cdn.jsdelivr.net/gh/highlightjs/cdn-release/build/styles/github.min.css">
<script src="https://cdn.jsdelivr.net/gh/highlightjs/cdn-release/build/highlight.min.js"></script>
```

### 2.3 组件设计

#### 2.3.1 MarkdownPreview 组件

```javascript
// js/components/MarkdownPreview.js
class MarkdownPreview {
  constructor(container, options = {}) {
    this.container = typeof container === 'string' 
      ? document.querySelector(container) 
      : container;
    
    this.options = {
      highlight: true,        // 是否代码高亮
      lineNumbers: false,     // 是否显示行号
      sanitize: false,        // 是否过滤 HTML
      breaks: true,           // 是否支持换行
      ...options
    };
    
    this._initMarked();
  }
  
  _initMarked() {
    // 配置 marked
    marked.setOptions({
      breaks: this.options.breaks,
      gfm: true,              // GitHub Flavored Markdown
      highlight: (code, lang) => {
        if (this.options.highlight && hljs) {
          if (lang && hljs.getLanguage(lang)) {
            return hljs.highlight(code, { language: lang }).value;
          }
          return hljs.highlightAuto(code).value;
        }
        return code;
      }
    });
  }
  
  render(markdown) {
    if (!markdown) {
      this.container.innerHTML = '<p class="empty">暂无内容</p>';
      return;
    }
    
    const html = marked.parse(markdown);
    this.container.innerHTML = html;
    
    // 添加样式类
    this.container.classList.add('markdown-body');
  }
  
  clear() {
    this.container.innerHTML = '';
    this.container.classList.remove('markdown-body');
  }
  
  export(format = 'html') {
    if (format === 'html') {
      return this.container.innerHTML;
    }
    // 可扩展导出为 PDF 等
  }
}
```

## 3. 功能二：大纲上传

### 3.1 需求描述

- 支持上传知识点大纲文档
- 解析文档生成导航栏内容
- 支持多种格式：JSON、Markdown、CSV

### 3.2 技术方案

**支持格式：**

| 格式 | 说明 | 示例 |
|------|------|------|
| JSON | 标准大纲格式 | `{"part": "...", "chapters": [...]}` |
| Markdown | 层级标题格式 | `# 部分\n## 章节\n### 知识点` |
| CSV | 表格格式 | `part,chapter,kp_id,name,desc` |

### 3.3 组件设计

#### 3.3.1 OutlineUploader 组件

```javascript
// js/components/OutlineUploader.js
class OutlineUploader {
  constructor(options = {}) {
    this.options = {
      accept: '.json,.md,.csv',
      maxSize: 10 * 1024 * 1024,  // 10MB
      ...options
    };
  }
  
  async upload(file) {
    // 1. 验证文件
    if (!this._validateFile(file)) {
      throw new Error('文件格式不支持或文件过大');
    }
    
    // 2. 解析文件
    const content = await this._readFile(file);
    const outline = this._parseContent(content, file.name);
    
    // 3. 上传到后端
    const formData = new FormData();
    formData.append('file', file);
    formData.append('name', file.name);
    
    const response = await fetch('/api/outline/upload', {
      method: 'POST',
      body: formData
    });
    
    return await response.json();
  }
  
  _validateFile(file) {
    const ext = file.name.split('.').pop().toLowerCase();
    const validExts = ['json', 'md', 'csv'];
    return validExts.includes(ext) && file.size <= this.options.maxSize;
  }
  
  async _readFile(file) {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = (e) => resolve(e.target.result);
      reader.onerror = reject;
      reader.readAsText(file);
    });
  }
  
  _parseContent(content, filename) {
    const ext = filename.split('.').pop().toLowerCase();
    
    switch (ext) {
      case 'json':
        return this._parseJSON(content);
      case 'md':
        return this._parseMarkdown(content);
      case 'csv':
        return this._parseCSV(content);
      default:
        throw new Error('Unsupported format');
    }
  }
  
  _parseJSON(content) {
    return JSON.parse(content);
  }
  
  _parseMarkdown(content) {
    // 解析 Markdown 层级结构
    const lines = content.split('\n');
    const outline = { chapters: [] };
    let currentChapter = null;
    let currentKp = null;
    
    for (const line of lines) {
      if (line.startsWith('# ')) {
        // 部分
        outline.part = line.substring(2).trim();
      } else if (line.startsWith('## ')) {
        // 章节
        currentChapter = { name: line.substring(3).trim(), kps: [] };
        outline.chapters.push(currentChapter);
      } else if (line.startsWith('### ')) {
        // 知识点
        currentKp = { name: line.substring(4).trim(), desc: '' };
        if (currentChapter) {
          currentChapter.kps.push(currentKp);
        }
      } else if (line.trim() && currentKp) {
        // 描述
        currentKp.desc += line.trim() + ' ';
      }
    }
    
    return outline;
  }
  
  _parseCSV(content) {
    const lines = content.split('\n');
    const headers = lines[0].split(',').map(h => h.trim());
    const outline = { chapters: [] };
    
    const chapterMap = new Map();
    
    for (let i = 1; i < lines.length; i++) {
      const values = lines[i].split(',').map(v => v.trim());
      const row = {};
      headers.forEach((h, idx) => row[h] = values[idx]);
      
      // 获取或创建章节
      if (!chapterMap.has(row.chapter)) {
        const chapter = { name: row.chapter, kps: [] };
        outline.chapters.push(chapter);
        chapterMap.set(row.chapter, chapter);
      }
      
      // 添加知识点
      const chapter = chapterMap.get(row.chapter);
      chapter.kps.push({
        id: row.kp_id,
        name: row.name,
        desc: row.desc || ''
      });
    }
    
    return outline;
  }
}
```

```javascript
router.post('/upload', upload.single('file'), async (req, res) => {

    const file = req.file;
```

## 4. 功能三：执行方案选择

### 4.1 需求描述

- 可以选择不同的 Agent 执行方案
- 支持预设方案：默认、快速、高质量
- 支持自定义方案

### 4.2 技术方案

**预设方案：**

| 方案 | 步骤 | 说明 |
|------|------|------|
| 默认 | planner → retriever → generator → validator | 标准4步流程 |
| 快速 | planner → retriever → generator | 跳过验证 |
| 高质量 | planner → retriever → generator → reviewer → validator | 增加审查 |

### 4.3 组件设计

#### 4.3.1 WorkflowSelector 组件

```javascript
// js/components/WorkflowSelector.js
class WorkflowSelector {
  constructor(container) {
    this.container = typeof container === 'string'
      ? document.querySelector(container)
      : container;
    this.workflows = [];
    this.selected = null;
  }
  
  async load() {
    const response = await fetch('/api/workflow/list');
    const result = await response.json();
    
    if (result.success) {
      this.workflows = result.data;
      this.render();
    }
  }
  
  render() {
    const html = `
      <div class="workflow-selector">
        <label>执行方案</label>
        <select id="workflowSelect" class="form-select">
          ${this.workflows.map(w => `
            <option value="${w.id}" ${w.is_default ? 'selected' : ''}>
              ${w.name} - ${w.description}
            </option>
          `).join('')}
        </select>
        <div class="workflow-steps" id="workflowSteps">
          ${this._renderSteps(this.workflows.find(w => w.is_default))}
        </div>
      </div>
    `;
    
    this.container.innerHTML = html;
    
    // 绑定事件
    document.getElementById('workflowSelect').addEventListener('change', (e) => {
      this._onSelect(e.target.value);
    });
  }
  
  _renderSteps(workflow) {
    if (!workflow) return '';
    
    return `
      <div class="steps-preview">
        ${workflow.steps.map((step, i) => `
          <span class="step-badge">${i + 1}. ${step.name}</span>
          ${i < workflow.steps.length - 1 ? '<span class="step-arrow">→</span>' : ''}
        `).join('')}
      </div>
    `;
  }
  
  _onSelect(workflowId) {
    const workflow = this.workflows.find(w => w.id === workflowId);
    this.selected = workflow;
    
    document.getElementById('workflowSteps').innerHTML = this._renderSteps(workflow);
  }
  
  getSelected() {
    return this.selected;
  }
}
```

## 5. 功能四：文档查看

### 5.1 需求描述

- 在前端查看生成的文档
- 支持 Markdown 渲染
- 支持文档列表、搜索

### 5.2 技术方案

**后端 API：**
- `GET /api/document/list`：获取文档列表
- `GET /api/document/:id`：获取文档详情
- `GET /api/document/:id/content`：获取文档内容

### 5.3 组件设计

#### 5.3.1 DocumentViewer 组件

```javascript
// js/components/DocumentViewer.js
class DocumentViewer {
  constructor(container) {
    this.container = typeof container === 'string'
      ? document.querySelector(container)
      : container;
    this.documents = [];
    this.currentDoc = null;
    this.preview = new MarkdownPreview('#docPreview');
  }
  
  async loadDocuments() {
    const response = await fetch('/api/document/list');
    const result = await response.json();
    
    if (result.success) {
      this.documents = result.data.items;
      this.renderList();
    }
  }
  
  renderList() {
    const html = `
      <div class="document-list">
        <div class="doc-header">
          <h3>生成的文档</h3>
          <input type="text" id="docSearch" placeholder="搜索文档...">
        </div>
        <div class="doc-items">
          ${this.documents.map(doc => `
            <div class="doc-item" data-id="${doc.id}">
              <div class="doc-title">${doc.title}</div>
              <div class="doc-meta">
                <span>${doc.knowledge_point}</span>
                <span>${formatDate(doc.created_at)}</span>
              </div>
            </div>
          `).join('')}
        </div>
      </div>
    `;
    
    this.container.innerHTML = html;
    
    // 绑定事件
    document.querySelectorAll('.doc-item').forEach(item => {
      item.addEventListener('click', () => {
        this._onSelectDoc(item.dataset.id);
      });
    });
  }
  
  async _onSelectDoc(docId) {
    const response = await fetch(`/api/document/${docId}/content`);
    const result = await response.json();
    
    if (result.success) {
      this.currentDoc = result.data;
      this.renderContent();
    }
  }
  
  renderContent() {
    if (!this.currentDoc) return;
    
    const html = `
      <div class="document-viewer">
        <div class="doc-header">
          <h2>${this.currentDoc.title}</h2>
          <div class="doc-actions">
            <button onclick="downloadDoc('${this.currentDoc.id}')">📥 下载</button>
            <button onclick="closeDocViewer()">✕ 关闭</button>
          </div>
        </div>
        <div id="docPreview" class="markdown-body"></div>
      </div>
    `;
    
    this.container.innerHTML = html;
    this.preview.render(this.currentDoc.content);
  }
}
```

## 9. 功能增强：提示词编辑模式

### 9.1 需求描述

- 生成的提示词需要支持在两种模式下编辑：
  1. **原始模式**：直接编辑 Markdown 源码
  2. **预览模式**：在渲染后的 HTML 上编辑（所见即所得）
- 两种模式可以实时切换，内容保持同步

### 9.2 设计方案

#### 9.2.1 三种模式

1. **查看模式（默认）**
   - 显示原始 Markdown 文本（outputBox）
   - 只读，不可编辑
   - 可以切换到编辑模式或预览模式

2. **编辑模式**
   - 显示 textarea，可以编辑 Markdown 源码
   - 实时语法高亮（可选）
   - 可以切换到查看模式或预览模式

3. **预览模式**
   - 显示渲染后的 HTML（markdownPreview）
   - 支持 contenteditable，可以直接编辑
   - 编辑后自动转换回 Markdown 格式
   - 可以切换到查看模式或编辑模式

#### 9.2.2 模式切换逻辑

```
查看模式 ←→ 编辑模式（textarea）
查看模式 ←→ 预览模式（contenteditable HTML）
编辑模式 ←→ 预览模式（需要转换）
```

#### 9.2.3 内容同步

- 从编辑模式切换到预览模式：
  1. 读取 textarea 中的 Markdown
  2. 使用 marked.js 渲染为 HTML
  3. 设置 contenteditable="true"
  
- 从预览模式切换到编辑模式：
  1. 读取 contenteditable 中的 HTML
  2. 使用 turndown.js 转换回 Markdown
  3. 填充到 textarea

- 从任意模式切换到查看模式：
  1. 保存当前内容到 outputBox

### 9.3 技术实现

#### 9.3.1 依赖库

- **marked.js**：Markdown → HTML（已集成）
- **turndown.js**：HTML → Markdown（需要新增）

#### 9.3.3 核心函数

```javascript
// 获取当前内容（Markdown 格式）
function getCurrentContent() {
  if (isEditing) {
    return document.getElementById('outputTextarea').value;
  } else if (isPreviewing) {
    const html = document.getElementById('markdownPreview').innerHTML;
    const turndown = new TurndownService();
    return turndown.turndown(html);
  } else {
    return document.getElementById('outputBox').textContent;
  }
}

// 设置内容
function setContent(markdown) {
  document.getElementById('outputBox').textContent = markdown;
  
  if (isEditing) {
    document.getElementById('outputTextarea').value = markdown;
  }
  
  if (isPreviewing) {
    const html = marked.parse(markdown);
    const preview = document.getElementById('markdownPreview');
    preview.innerHTML = html;
    preview.contentEditable = 'true';
  }
}

// 切换到编辑模式
function switchToEditMode() {
  const markdown = getCurrentContent();
  isEditing = true;
  isPreviewing = false;
  
  document.getElementById('outputBox').style.display = 'none';
  document.getElementById('markdownPreview').style.display = 'none';
  document.getElementById('outputTextarea').style.display = 'block';
  document.getElementById('outputTextarea').value = markdown;
  
  updateModeButtons();
}

// 切换到预览模式
function switchToPreviewMode() {
  const markdown = getCurrentContent();
  const html = marked.parse(markdown);
  
  isEditing = false;
  isPreviewing = true;
  
  document.getElementById('outputBox').style.display = 'none';
  document.getElementById('outputTextarea').style.display = 'none';
  const preview = document.getElementById('markdownPreview');
  preview.style.display = 'block';
  preview.innerHTML = html;
  preview.contentEditable = 'true';
  
  updateModeButtons();
}

// 切换到查看模式
function switchToViewMode() {
  const markdown = getCurrentContent();
  
  isEditing = false;
  isPreviewing = false;
  
  document.getElementById('outputTextarea').style.display = 'none';
  document.getElementById('markdownPreview').style.display = 'none';
  const box = document.getElementById('outputBox');
  box.style.display = 'block';
  box.textContent = markdown;
  
  updateModeButtons();
}
```
