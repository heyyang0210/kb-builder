# 18-文档预处理与大纲管理设计

## 1. 概述

本文档描述两个核心功能的设计：
1. **文档预处理**：上传的原始文档经过预处理，转换为系统标准输入格式
2. **大纲管理增强**：支持删除已上传大纲 + 从大纲生成提示词

## 2. 功能一：文档预处理

### 2.1 需求背景

用户上传的文档格式多样（Word、PDF、Markdown、纯文本等），需要统一预处理为系统期望的标准格式后再进行后续处理。

### 2.2 标准输入格式

系统期望的标准文档格式为结构化 JSON：

```json
{
  "title": "文档标题",
  "source_file": "原始文件名",
  "source_type": "md|pdf|docx|txt",
  "processed_at": "2026-07-07T10:00:00Z",
  "content": {
    "sections": [
      {
        "level": 1,
        "title": "一级标题",
        "content": "段落内容",
        "children": []
      }
    ]
  },
  "metadata": {
    "word_count": 1000,
    "section_count": 5,
    "has_code_blocks": true,
    "has_tables": true
  }
}
```

### 2.3 预处理流程

```
┌─────────────┐     ┌──────────────┐     ┌─────────────┐     ┌──────────────┐
│  上传原始文档 │ ──▶ │  格式检测与解析 │ ──▶ │  结构化提取   │ ──▶ │  生成标准格式  │
└─────────────┘     └──────────────┘     └─────────────┘     └──────────────┘
                           │                    │                    │
                           ▼                    ▼                    ▼
                    ┌──────────────┐     ┌─────────────┐     ┌──────────────┐
                    │ - Markdown   │     │ - 标题层级   │     │ - JSON 输出   │
                    │ - PDF        │     │ - 段落内容   │     │ - 元数据统计   │
                    │ - Word       │     │ - 代码块     │     │ - 预览支持     │
                    │ - 纯文本     │     │ - 表格       │     │              │
                    └──────────────┘     └─────────────┘     └──────────────┘
```

```javascript
// routes/document.js

// POST /api/document/preprocess
router.post('/preprocess', upload.single('file'), async (req, res) => {
  try {
    if (!req.file) {
      return res.status(400).json({ success: false, message: '未选择文件' });
    }

    const file = req.file;
    const ext = path.extname(file.originalname).toLowerCase();
    
    // 1. 根据文件类型解析
    let rawContent;
    switch (ext) {
      case '.md':
        rawContent = parseMarkdown(file.path);
        break;
      case '.txt':
        rawContent = parsePlainText(file.path);
        break;
      case '.pdf':
        rawContent = await parsePDF(file.path);
        break;
      case '.docx':
        rawContent = await parseDocx(file.path);
        break;
      default:
        return res.status(400).json({ success: false, message: '不支持的文件格式' });
    }

    // 2. 生成标准格式
    const docId = `doc_${Date.now()}`;
    const standardDoc = {
      id: docId,
      title: rawContent.title || path.basename(file.originalname, ext),
      source_file: file.originalname,
      source_type: ext.slice(1),
      processed_at: new Date().toISOString(),
      content: rawContent.content,
      metadata: {
        word_count: countWords(rawContent.content),
        section_count: countSections(rawContent.content),
        has_code_blocks: hasCodeBlocks(rawContent.content),
        has_tables: hasTables(rawContent.content)
      }
    };

    // 3. 保存元数据
    saveDocumentMetadata(standardDoc);

    // 4. 保存原始文件（用于后续处理）
    const savedPath = path.join(UPLOAD_DIR, `${docId}${ext}`);
    fs.copyFileSync(file.path, savedPath);

    res.json({
      success: true,
      data: standardDoc
    });

  } catch (err) {
    console.error('Document preprocess error:', err);
    res.status(500).json({ success: false, message: err.message });
  }
});

// 解析 Markdown 文件
function parseMarkdown(filePath) {
  const content = fs.readFileSync(filePath, 'utf-8');
  const lines = content.split('\n');
  
  const result = {
    title: '',
    sections: []
  };
  
  let currentSection = null;
  let sectionStack = [result.sections]; // 用于处理嵌套层级
  
  for (const line of lines) {
    // 检测标题
    const headingMatch = line.match(/^(#{1,6})\s+(.+)$/);
    if (headingMatch) {
      const level = headingMatch[1].length;
      const title = headingMatch[2].trim();
      
      if (level === 1 && !result.title) {
        result.title = title;
        continue;
      }
      
      const section = {
        level: level,
        title: title,
        content: '',
        children: []
      };
      
      // 找到合适的父级
      while (sectionStack.length > level) {
        sectionStack.pop();
      }
      
      const parent = sectionStack[sectionStack.length - 1];
      parent.push(section);
      sectionStack.push(section.children);
      currentSection = section;
      continue;
    }
    
    // 累积内容到当前节
    if (currentSection) {
      currentSection.content += line + '\n';
    }
  }
  
  return { title: result.title, content: result.sections };
}
```

```javascript
// routes/outline.js - DELETE /api/outline/:id（已存在）
router.delete('/:id', (req, res) => {
  try {
    const metadataPath = path.join(__dirname, '..', 'outlines', 'metadata.json');
    
    if (!fs.existsSync(metadataPath)) {
      return res.status(404).json({ success: false, message: '大纲不存在' });
    }
    
    const metadata = JSON.parse(fs.readFileSync(metadataPath, 'utf-8'));
    const outlineIndex = metadata.outlines.findIndex(o => o.id === req.params.id);
    
    if (outlineIndex === -1) {
      return res.status(404).json({ success: false, message: '大纲不存在' });
    }
    
    const outline = metadata.outlines[outlineIndex];
    
    // 删除文件
    const filePath = path.join(__dirname, '..', 'outlines', outline.file);
    if (fs.existsSync(filePath)) {
      fs.unlinkSync(filePath);
    }
    
    // 从元数据中移除
    metadata.outlines.splice(outlineIndex, 1);
    fs.writeFileSync(metadataPath, JSON.stringify(metadata, null, 2));
    
    res.json({ success: true, message: '大纲已删除' });
  } catch (err) {
    res.status(500).json({ success: false, message: err.message });
  }
});
```

#### 后端：大纲提示词生成

```javascript
// routes/outline.js - POST /api/outline/generate-prompts
router.post('/generate-prompts', async (req, res) => {
  try {
    const { outline_id, knowledge_points, options = {} } = req.body;
    
    if (!outline_id || !knowledge_points || knowledge_points.length === 0) {
      return res.status(400).json({ 
        success: false, 
        message: '缺少必要参数: outline_id, knowledge_points' 
      });
    }
    
    // 1. 获取大纲数据
    const metadataPath = path.join(__dirname, '..', 'outlines', 'metadata.json');
    const metadata = JSON.parse(fs.readFileSync(metadataPath, 'utf-8'));
    const outline = metadata.outlines.find(o => o.id === outline_id);
    
    if (!outline || !outline.content) {
      return res.status(404).json({ success: false, message: '大纲不存在或未解析' });
    }
    
    // 2. 提取知识点详情
    const kpDetails = extractKnowledgePoints(outline.content, knowledge_points);
    
    // 3. 生成提示词
    const prompts = [];
    for (const kp of kpDetails) {
      const promptData = {
        name: kp.name,
        part: kp.part,
        chapter: kp.chapter,
        desc: kp.desc,
        type: detectType(kp.name, kp.desc),
        targetDb: options.target_db || 'YashanDB',
        refMcp: '',
        refDesign: '',
        refOracle: '',
        refTest: ''
      };
      
      const result = assemblePrompt(promptData);
      prompts.push({
        kp_id: kp.id,
        kp_name: kp.name,
        prompt: result.prompt,
        skill_file: result.skillFile,
        template_file: result.templateFile
      });
    }
    
    // 4. 合并输出（可选）
    let mergedPrompt = null;
    if (options.merge_output) {
      mergedPrompt = prompts.map(p => 
        `# ${p.kp_name}\n\n${p.prompt}\n\n---\n\n`
      ).join('\n');
    }
    
    res.json({
      success: true,
      data: {
        prompts,
        merged_prompt: mergedPrompt
      }
    });
    
  } catch (err) {
    console.error('Generate prompts error:', err);
    res.status(500).json({ success: false, message: err.message });
  }
});

// 从大纲内容中提取知识点详情
function extractKnowledgePoints(outlineContent, selectedKps) {
  const result = [];
  const parts = outlineContent.parts || [];
  
  for (const selected of selectedKps) {
    const part = parts.find(p => p.part === selected.part);
    if (!part) continue;
    
    const chapter = part.chapters.find(c => c.name === selected.chapter);
    if (!chapter) continue;
    
    const kp = chapter.kps.find(k => k.id === selected.kp_id);
    if (!kp) continue;
    
    result.push({
      id: kp.id,
      name: kp.name,
      desc: kp.desc,
      part: part.part,
      chapter: chapter.name
    });
  }
  
  return result;
}
```

```javascript
function detectOutlineFormat(content) {
  const outlinePatterns = [
    /^##\s+[一二三四五六七八九十]+、/m,
    /^-\s+\*\*\d+\.\d+\s+/m,
    /^-\s+\d+\.\d+\.\d+\s+[⭐★]/m
  ];
  let matchCount = 0;
  for (const pattern of outlinePatterns) {
    if (pattern.test(content)) matchCount++;
  }
  return matchCount >= 2;
}
```

```javascript
function selectKp(kpId) {
  const parts = kpId.split('-');
  const domain = parts[0]; // 'db', 'biz', or 'up'
  
  if (domain === 'up') {
    // 上传的大纲：up-oi-pi-ci-ki
    const oi = parseInt(parts[1]); // outline index
    const pi = parseInt(parts[2]); // part index
    const ci = parseInt(parts[3]); // chapter index
    const ki = parseInt(parts[4]); // kp index
    
    const outline = uploadedOutlines[oi];
    const part = outline.parts[pi];
    const chapter = part.chapters[ci];
    const kp = chapter.kps[ki];
    
    // 填充表单
    document.getElementById('kpPart').value = part.part;
    document.getElementById('kpName').value = kp.name;
    document.getElementById('kpDesc').value = kp.desc || '';
    document.getElementById('kpType').value = detectType(kp.name, kp.desc);
  }
}
```
