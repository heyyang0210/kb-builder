##   [1. Overview（概述）](https://conf.yasdb.com/pages/viewpage.action?pageId=89094607#1-overview%E6%A6%82%E8%BF%B0)  

集群下支持导入导出。集群和单机的部署形式，对工具/驱动而言是透明的，因此，只要集群环境下支持的对象，导入导出原则上都是支持的。先对集群支持的对象进行以下的整理。

--range分区
drop table if exists rangeTable1;
create table rangeTable1(a int, b int) partition by range(a, b )
(
partition p1 values less than(1,2),
partition p2 values less than(10,20)
);
insert into rangeTable1 values(1,1);
commit;
--range 分区 tablespace分区不同
drop table if exists rangeTable2;
create table rangeTable2(a int, b varchar(4000)) tablespace users pctfree 10 initrans 8 partition by range(a, b)
(partition p1 values less than(1,'a') tablespace system pctfree 5 initrans 5 maxtrans 100,
partition p2 values less than(10, 'c'),
partition p3 values less than(MAXVALUE, MAXVALUE)
);
insert into rangeTable2 values(1,'a');
commit;
--range分区自动生成分区名称partition不带名字
drop table if exists rangeTable3;
create table rangeTable3(a int, b varchar(4000))
partition by range(a, b)
(
partition values less than(1,'a') tablespace system pctfree 5 initrans 5 maxtrans 100,
partition values less than(10, 'c') ,
partition values less than(MAXVALUE, MAXVALUE)
);
--list分区
drop table if exists listTable1;
create table listTable1(a int, b int) partition by list(a, b)
(
partition p1 values (1,2),
partition p2 values (2,3),
partition p3 values (default)
);
insert into listTable1 values(1,1);
insert into listTable1 values(2,2);
commit;

--list分区,不指定分区名字
drop table if exists listTable2;
create table listTable2(a int, b int) partition by list(a, b)
(
partition  values (1,2),
partition  values (2,3),
partition  values (default)
);
insert into listTable2 values(1,1);
insert into listTable2 values(2,2);
commit;
--hash分区-自动生成分区，导出完整的分区信息
drop table if exists hashQuantitytable1;
create table hashQuantitytable1(a int, b int) partition by hash(a) partitions 5 store in (system,users);
insert into hashQuantitytable1 values (1,1);
insert into hashQuantitytable1 values (10,10);
commit;
--hash分区用户定义分区个数
drop table if exists hashQuantitytable2;
create table hashQuantitytable2(a int, b int) partition by hash(a)
(
	partition p1 tablespace users,
	partition p2 tablespace system
);
insert into hashQuantitytable2 values (1,1);
insert into hashQuantitytable2 values (10,10);
commit;
--intervel分区
drop table if exists interval_test;
drop table if exists  interval_test;
create table interval_test(a int)  partition by range(a) interval(2)(partition p1 values less than(5),partition p2 values less than(10));
insert into interval_test values (11);
insert into interval_test values (15);
commit;


--index：分区、非分区
drop table  if exists student1;
create  table student1 (id number,stu_no varchar2(100))
partition by RANGE (id)
(
partition p1 values less than (100) tablespace users,
partition p2 values less than (200) tablespace users,
partition p3 values less than (maxvalue) tablespace users
);
create index stu_prefix_index on student1(id) tablespace system local;
create  index stu_noprefix_index on student1(stu_no) local;
drop table if exists  student2;
create  table student2 (id number,stu_no varchar2(100))
partition by RANGE (id)
(
partition p1 values less than (100) tablespace users,
partition p2 values less than (200) tablespace users,
partition p3 values less than (maxvalue) tablespace users
);
create index stu_prefix_index2 on student2(id) tablespace system INITRANS 2 pctfree 2  local (PARTITION P1 TABLESPACE default INITRANS 3, PARTITION INITRANS 2 pctfree 2 tablespace users,PARTITION );
--store in属性应用在hash分区
drop table if exists hashTable;
create table hashTable(a int, b int) partition by hash(a) partitions 2 store in (system,users);
create index hashTable_index on hashTable(a) tablespace system local store in(users,system);





--约束：check、UNIQUE、主键、外键

ALTER TABLE table_export add CONSTRAINT "bb" UNIQUE(id4);
--索引

CREATE INDEX index_export2 on table_export(id2);
--主键
ALTER TABLE table_export ADD CONSTRAINT PRIMARY_export PRIMARY KEY(id1);
--外键
ALTER TABLE table_foregin ADD CONSTRAINT FOREIGN_export FOREIGN KEY(id1) REFERENCES table_export(id1);
