# 一、功能

|功能|描述|
|---|---|
|跨一级分区更新|update的时候修改一级分区键|
|跨二级分区更新|update的时候修改二级分区键|


# 二、用例

首先enable row movement

alter table table_name enable row movement;

|分区类型|测试场景|用例|输出结果|
|---|---|---|---|
|hash-hash,  
,  
|跨一级分区更新|drop table hh_composite;    
  create table hh_composite(a int, b varchar(10))    
  partition by hash(a)    
  subpartition by hash(b)    
  subpartitions 8    
  (partition p1, partition p2) enable row movement;    
  insert into hh_composite values(1, '1');    
  select * from hh_composite partition(p2); --查到一行    
  update hh_composite set a=2 where a=1; --从p2 update到p1    
  select * from hh_composite partition(p2); --查不到    
  select * from hh_composite partition(p1); --查到一行    
  select * from hh_composite where a=2; --查到一行    
  select * from hh_composite where a=1; --查不到|成功。|
||跨二级分区更新|drop table hh_composite;    
  create table hh_composite(a int, b varchar(10))    
  partition by hash(a)    
  subpartition by hash(b)    
  subpartition template(subpartition subp1, subpartition subp2)    
  (partition p1,partition p2) enable row movement;    
  insert into hh_composite values(1, '13');    
  select * from hh_composite subpartition(p2_subp1); --查到一行    
  update hh_composite set b='1' where a=1; --从p2_subp1 update到p2_subp2    
  select * from hh_composite subpartition(p2_subp1); --查不到    
  select * from hh_composite subpartition(p2_subp2); --查到一行|成功。|
