# 1.概述

- 提供几何图形的xmin, ymin, xmax, ymax，另外如果有需求还可以提供zmin,zmax,mmin,mmax
- 在解析wkb时，将其转换的过程中处理其二维边界框的解析
- 内部函数逻辑


  


# 2.接口

```
static inline void calcBBox(BBox4D* box, uint32_t coordinateIndex, double coordinate);

```

  


# 3.详细设计

bbox类型：

```
typedef struct StBBox4D {
    double xmin;
    double xmax;
    double ymin;
    double ymax;
    double zmin;
    double zmax;
    double mmin;
    double mmax;
} BBox4D;

```

解析从wkb到我们自研的StGeometry的转换：

```
YspiResult piParseWkb(YspiHandle hExec, uint8_t* wkb, uint32_t wkbSize, Geometry** geom);

```

无论任何图形，最后都会解析其最终的坐标，比如集合类型，最终都是点：

```
static YspiResult parseWkbCoordinate(YspiHandle hExec, WkbParseAssist* assist, double* coordinates)
{
    for (uint32_t i = 0; i &lt; assist-&gt;dims; i++) {
        if (piWkbReadDouble(hExec, assist, &amp;coordinates[i]) != YSPI_SUCCESS) {
            return YSPI_ERROR;
        }
    }
    return YSPI_SUCCESS;
}

```

所以不需要在解析完成后单独做bbox的计算，而是在解析的过程中，通过解析具体坐标时不断更新bbox的坐标最值即可：

```
static YspiResult parseWkbCoordinate(YspiHandle hExec, WkbParseAssist* assist, double* coordinates)
{
    for (uint32_t i = 0; i &lt; assist-&gt;dims; i++) {
        if (piWkbReadDouble(hExec, assist, &amp;coordinates[i]) != YSPI_SUCCESS) {
            return YSPI_ERROR;
        }
        if (calcBBox(&amp;assist-&gt;bbox, i, coordinates[i]) != YSPI_SUCCESS) {
            return YSPI_ERROR;
        }
    }
    return YSPI_SUCCESS;
}

```

```
static inline bool isValidBBox(BBox4D* box, Geometry* geom) 
{
    if (
        box-&gt;xmin &gt;= box-&gt;xmax ||
        box-&gt;ymin &gt;= box-&gt;ymax ||
        (geom-&gt;hasZ &amp;&amp; box-&gt;zmin &gt;= box-&gt;zmax) ||
        (geom-&gt;hasM &amp;&amp; box-&gt;mmin &gt;= box-&gt;mmax)
    ) {
        return false;
    }

    return true;
}

```

## 与postgis规格差异

|~~场景~~|~~postgis~~|~~自研~~|
|---|---|---|
|~~外包框存在NaN~~|~~支持~~|~~不支持~~|


2023.05.24修改：由于现在外包框与postgis对齐，已支持Nan的输入，也就是说，外包框中会存在Nan等特殊值，将此类特殊值的外包框作为参数调用geos的接口并不一定保证输出结果的有效性。

规格差异：

|场景|postgis|自研|
|---|---|---|
|外包框支持Inf、-Inf的输入|不支持|支持|


postgis不支持inf、-inf的输出，但是我们支持，同样的，inf、-inf这样的特殊值输入并不保证输出结果的有效性

# 4.测试用例

```
-- 输出应为空
select st_astext(st_clipbybox2d(st_geomfromtext('polygon((2 0, 3 0, 3 2, 2 3, 0 2, 2 0))'), st_geomfromtext('multipoint(empty, (1 nan), (3 1), empty, (4 nan))')), 0) from dual;
select st_astext(st_clipbybox2d(st_geomfromtext('polygon((2 0, 3 0, 3 2, 2 3, 0 2, 2 0))'), st_geomfromtext('point(nan 1)')), 0) from dual;
select st_astext(st_clipbybox2d(st_geomfromtext('polygon((2 0, 3 0, 3 2, 2 3, 0 2, 2 0))'), st_geomfromtext('point(empty)')), 0) from dual;
-- 正常输出
select st_astext(st_clipbybox2d(st_geomfromtext('polygon((2 0, 3 0, 3 2, 2 3, 0 2, 2 0))'), st_geomfromtext('multipoint(empty, (1 2), (3 4), empty, (4 5))')), 0) from dual;


```
