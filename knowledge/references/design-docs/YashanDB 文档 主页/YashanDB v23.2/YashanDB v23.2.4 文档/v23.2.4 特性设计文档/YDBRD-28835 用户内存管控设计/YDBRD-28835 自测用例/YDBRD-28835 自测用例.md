前置操作

```
-- 创建资源组
exec dbms_resource_manager.create_consumer_group('resgroup', 'group for mem resource test');
select consumer_group_id, consumer_group, comments from DBA_RSRC_CONSUMER_GROUPS where consumer_group = upper('resgroup');
-- 创建计划
exec dbms_resource_manager.create_plan('resplan', 'plan for mem resource test');
select plan_id, plan, num_plan_directives, comments from DBA_RSRC_plans where plan = upper('resplan');

-- 查看指令
select plan, group_or_subplan, spa_limit, session_spa_limit, execution_queue_timeout from dba_rsrc_plan_directives where plan = upper('resplan');

```
