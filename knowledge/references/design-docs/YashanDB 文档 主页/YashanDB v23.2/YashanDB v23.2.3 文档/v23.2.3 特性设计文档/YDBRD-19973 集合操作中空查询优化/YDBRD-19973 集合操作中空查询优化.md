  #YDBRD-19973 集合操作中空查询优化-Union/Union All

##   [1. Overview（概述）](http://cod-conf.sics.com/pages/viewpage.action?pageId=72797254#1-overview%E6%A6%82%E8%BF%B0)  

集合操作，部分子语句的filter恒为false，按照最优计划来说，当前子语句可以优化掉，不需要执行。可以在计划层面上将这个语句优化掉，从而可以节省执行时间。

##   [2. Features（功能特性）](http://cod-conf.sics.com/pages/viewpage.action?pageId=72797254#2-features%E5%8A%9F%E8%83%BD%E7%89%B9%E6%80%A7)  

集合操作，优化空查询，在计划层不会生成对应的执行计划。

##   [3. Interfaces（接口）](http://cod-conf.sics.com/pages/viewpage.action?pageId=72797254#3-interfaces%E6%8E%A5%E5%8F%A3)  

集合操作，对外SQL接口无变化。

##   [4. Limitations（功能限制）](http://cod-conf.sics.com/pages/viewpage.action?pageId=72797254#4-limitations%E5%8A%9F%E8%83%BD%E9%99%90%E5%88%B6)  

|  
|约束|备注|
|---|---|---|
|1|update，insert，delete，merge先不支持，所有优化针对 select|  
|
|2|not exists (全量数据) 和 not in (全量数据)的anti jion恒false暂时不支持，后续优化支持|  
|
|3|单机limit 0暂时不支持绑定参数，绑定参数动态值不会判断为恒false|  
|
|4|单机场景rownum表达式恒flase不支持，后续优化支持|  
|
|5|union/union all左右都为恒false场景暂时不优化，等后续需求修改删除在静态改写步骤做优化|  
|

static CodResult preOptmzNullCmpFilters(AnlStmt* stmt, MemoryContext* mctx, FilterSets* filterSets)
{
    if (isObjectArrayEmpty(filterSets-&gt;cmpFilters)) {
        return COD_SUCCESS;
    }

    CodUint32   i = 0;
    FilterNode* fNode = NULL;
    ExprNode*   rNode = NULL;
    ExprNode*   lNode = NULL;
    FilterNode* newFilter = NULL;
    while (i &lt; filterSets-&gt;cmpFilters-&gt;count) {
        fNode = (FilterNode*)objArrayGet(filterSets-&gt;cmpFilters, i);

        rNode = fNode-&gt;cmp.left-&gt;root;
        lNode = fNode-&gt;cmp.right-&gt;root;
        if ((isExprNodeConst(rNode) &amp;&amp; rNode-&gt;value.isNull) ||
            (isExprNodeConst(lNode) &amp;&amp; lNode-&gt;value.isNull)) {
            COD_CALL(makeBoolFilterNode(mctx, FILTER_FALSE, &amp;newFilter));
            COD_CALL(extractFalseFilter(filterSets, newFilter));
            objArrayDelete(filterSets-&gt;cmpFilters, i);
            continue;
        }
		
        // 新增是否恒false判断处理
        if (isStaticFilter(fNode)) {
            COD_CALL(cboOptmzStaticFilter(stmt, fNode));
            if (fNode-&gt;type == FILTER_FALSE) {
                COD_CALL(makeBoolFilterNode(mctx, FILTER_FALSE, &amp;newFilter));
                COD_CALL(extractFalseFilter(filterSets, newFilter));
                objArrayDelete(filterSets-&gt;cmpFilters, i);
                continue;
            } else if (fNode-&gt;type == FILTER_TRUE) {
                COD_CALL(makeBoolFilterNode(mctx, FILTER_TRUE, &amp;newFilter));
                COD_CALL(extractTrueFilter(filterSets, newFilter));
                objArrayDelete(filterSets-&gt;cmpFilters, i);
                continue;
            }
        }
        i = i + 1;
    }

    if (filterSets-&gt;cmpFilters-&gt;count == 0) {
        filterSets-&gt;cmpFilters = NULL;
    }
    return COD_SUCCESS;
}

static CodResult removeEmptyUnionMember(TransContext* transCtx, CboMemo* cboMemo, CboGroupExpr* srcGExpr)
{
    CboOperator* op = srcGExpr-&gt;cboOp;
    COD_ASSERT(isUnionAllOp(op) || isUnionOp(op));

    CboGroupExpr*  subGExpr;
    MemoryContext* mctx = transCtx-&gt;cboOpt-&gt;mctx;
    ObjectArray*   dstArray;

    COD_CALL(shallowCopyArray(mctx, srcGExpr-&gt;childGroups, &amp;dstArray, COD_TRUE));
    for (CodUint32 i = 0; i &lt; srcGExpr-&gt;childGroups-&gt;count; ++i) {
        subGExpr = (CboGroupExpr*)getChildLogiGExpr(srcGExpr, i);
        if (isEmptyDSGExpr(subGExpr)) {
            objArrayFastDelete(srcGExpr-&gt;childGroups, i);
            i--;
        }
    }

    if (srcGExpr-&gt;childGroups-&gt;count == 1) {
        subGExpr = (CboGroupExpr*)getChildLogiGExpr(srcGExpr, 0);
        // need change to insertGroup hy
        *srcGExpr-&gt;group = *subGExpr-&gt;group;
        srcGExpr-&gt;cboOp = subGExpr-&gt;cboOp;
        srcGExpr-&gt;childGroups = subGExpr-&gt;childGroups;
        srcGExpr-&gt;reqdProp = subGExpr-&gt;reqdProp;
    } else if (srcGExpr-&gt;childGroups-&gt;count == 0) {
        srcGExpr-&gt;childGroups = dstArray;
//        CboOperator* resultOp;
//        COD_CALL(makeResultFalse(mctx, &amp;resultOp));
//        COD_CALL(insertGroup(mctx, cboMemo, resultOp, &amp;srcGExpr-&gt;group, GROUP_EXPLORED, COD_FALSE));
//        srcGExpr-&gt;cboOp = resultOp;
//        srcGExpr-&gt;childGroups = dstArray;
    }

    return COD_SUCCESS;
}

CodResult txformLogiUnionSimplify(TransContext* transCtx, CboGroupExpr* srcGExpr, CboMemo* cboMemo, CboGroupExpr** dstGExpr)
{
    //rule 1: remove empty select from union/union all.
    COD_CALL(removeEmptyUnionMember(transCtx, cboMemo, srcGExpr));
    return COD_SUCCESS;
}
