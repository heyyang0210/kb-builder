typedef struct StGcsReleaseWorker {
    CodUint8         id;
    CodUint8         unused[3];
    CodThread        thread;
    AnkHandler*      handler;
    volatile CodBool workCompleted;
    volatile CodBool running;

typedef struct StGcsReleaseManager {
    SpinLock           lock;
    volatile CodUint32 workerCnt;
    GcsReleaseWorker   workers[GCS_RELEASE_WORKER_CNT];

####   [5.2 流程设计](#52-流程设计)  

- shutdown流程中释放全局页面资源时，创建最多8个线程并发进行资源释放
- 线程创建失败，利用已经创建的线程进行资源释放。如果一个线程都没有创建成功，利用shutdown线程做资源释放
- bufferChunk按照线程数进行切分，多线程并发释放资源
