一、对于代码中不再使用，参数状态变更为过时的参数：    
  **1.anr_param.c文件确认gParamItems是否需要新增相关过时参数宏：**

当前已有过时参数宏为：

```
#define DEPRE_NORMAL_PARAM_DEF(vId, vName, vDefault, vSetSysParam) [vId] = {\
    .section        = COD_TEXT_DEF(PARAM_SECTION_NAME),               \
    .unit           = COD_TEXT_DEF(PARAM_UNIT_NAME),                  \
    .name           = COD_TEXT_DEF(vName),                            \
    .id             = vId,                                            \
    .defaultValue   = COD_TEXT_DEF(vDefault),                         \
    .value          = COD_TEXT_BUF(gNormalParamBuffers[vId]),         \
    .bufSize        = PARAM_NORMAL_SIZE,                              \
    .isVisible      = COD_TRUE,                                       \
    .isDeprecated   = COD_TRUE,                                       \
    .setSysParam    = vSetSysParam,                                   \
    .setSesParam    = NULL,                                           \
    .getSesParam    = NULL,                                           \
}

#define DEPRE_INVIS_PARAM_DEF(vId, vName, vDefault, vSetSysParam) [vId] = {\
    .section        = COD_TEXT_DEF(PARAM_SECTION_NAME),              \
    .unit           = COD_TEXT_DEF(PARAM_UNIT_NAME),                 \
    .name           = COD_TEXT_DEF(vName),                           \
    .id             = vId,                                           \
    .defaultValue   = COD_TEXT_DEF(vDefault),                        \
    .value          = COD_TEXT_BUF(gNormalParamBuffers[vId]),        \
    .bufSize        = PARAM_NORMAL_SIZE,                             \
    .isVisible      = COD_FALSE,                                     \
    .isDeprecated   = COD_TRUE,                                      \
    .setSysParam    = vSetSysParam,                                  \
    .setSesParam    = NULL,                                          \
    .getSesParam    = NULL,                                          \
}
```

若当前宏不满足，则新增宏，新宏为对应原有宏的.isDeprecated 修改为COD_TRUE即可。

注意，对非隐藏参数新增过时参数宏，将影响到ha_regress测试框架（门禁），ha_regress使用参见：    [HA环境搭建，ha regress，ha_regress使用说明， - YashanDB - SICS-CoD Confluence (yasdb.com)](https://conf.yasdb.com/pages/viewpage.action?pageId=76907925)    ，

影响文件和内容见：    [add DEPRE_NORMAL_PARAM_DEF (!1385) · Merge requests · CoD-X / anchor_regress · GitLab (yasdb.com)](https://git.yasdb.com/cod-x/anchor_regress/-/merge_requests/1385/diffs)    ，

主要影响原因是因为用例会对anr_param.c文件的gParamItems数组根据用例的非隐藏参数宏和文档的配置参数.md进行比较，因此需要对用例也同样增加对新增的宏的比较。

**2.anr_param.c的gParamItems数组对参数的宏修改为对应过时参数宏，例：**

```
DEPRE_INVIS_PARAM_DEF(PARAM_ENABLE_DSTB_TRANS,    "_ENABLE_DISTRIBUTE_TRANSACTION",  "FALSE", cbpmEnableDstbTrans),
```

宏从原来的INVIS_PARAM_DEF变更为DEPRE_INVIS_PARAM_DEF。

**3.CodParamItem的setSysParam加上日志告警logDeprecatedParam，例：**

```
static inline CodVoid logDeprecatedParam(CodChar* name)
{
    COD_LOG_WARN(COD_SUCCESS,"[PARAM] parameter %s is deprecated, it will obsolete in later version", name);
}
static CodResult cbpmEnableDstbTrans(CodParamItem* item, CodText* value, CodParamScope scope)
{
    CHECK_SPFILE(scope);
    TRIM_WRAPPER(value);
    UPPER_VALUE(value);
    CodBool isEnable;
    COD_CALL(codText2Bool(value, &isEnable));
    logDeprecatedParam(item->name.str);
    return COD_SUCCESS;
}
```

如果setSesParam不为空，则也需要加上日志告警logDeprecatedParam。

### 二、对于改名的过时参数：

**1.anr_param.c文件确认gParamItems是否需要新增相关过时参数宏，同一；**

**2.anr_param.c的gParamItems数组，新参数除参数名和增加paramId外，建议相关接口名、函数名和变量名也修改为新参数相关，若默认值变更，也需要修改；**

**原参数的宏修改为对应过时参数宏，setSysParam、setSesParam、getSesParam和新参数用同一接口，例：**

```
原：
    NORMAL_PARAM_DEF(PARAM_DIN_RECONNECT_TIME,       "DIN_RECONNECT_TIME",   "5000",  cbpmDinHbTimeout),
    NORMAL_PARAM_DEF(PARAM_BROADCAST_GTS_TIME, "BROADCAST_GTS_TIME", "5", cbpmBroadcastGtsTime),

修改后：
    DEPRE_NORMAL_PARAM_DEF(PARAM_DIN_RECONNECT_TIME,       "DIN_RECONNECT_TIME",   "5000",  cbpmDinHbTimeout),
    NORMAL_PARAM_DEF(PARAM_DIN_HB_TIMEOUT, "DIN_HB_TIMEOUT", "15", cbpmDinHbTimeout),
    DEPRE_NORMAL_PARAM_DEF(PARAM_BROADCAST_GTS_TIME, "BROADCAST_GTS_TIME", "5", cbpmGtsBroadcastTime),
    NORMAL_PARAM_DEF(PARAM_GTS_BROADCAST_TIME, "GTS_BROADCAST_TIME", "5", cbpmGtsBroadcastTime),
```

**3.修改CodParamItem的setSysParam。过时参数打印日志告警logDeprecatedParam；在scope包含memory时，若参数未过时，更新实例的attr，或者参数过时但未配置新参数同时在做修改而不是启动，则先将旧参数值转换给新参数后再加载到实例的attr（注意，个别参数参数规则有变更，此处对于未配置新参数的过时参数，需匹配新参数规则，参考din_reconnect_time)，例：**

```
原：
static CodResult cbpmBroadcastGtsTime(CodParamItem* item, CodText* value, CodParamScope scope)
{
    TRIM_WRAPPER(value);
    UPPER_VALUE(value);
    CodUint32 val;
    COD_CALL(anrParamSizeConvAndCheck(&item->name, value, &val,
                                      (CodUint64)ANH_MIN_BROADCAST_GTS_TIME, (CodUint64)ANH_MAX_BROADCAST_GTS_TIME));
    if (COD_PARAM_SCOPE_CONTAIN(scope, COD_PARAM_SCOPE_MEMORY)) {
        AnlAttr* attr = anlGetAttr(gInstance->sql);
        attr->broadcastGtsTime = val;
    }

    return COD_SUCCESS;
}
修改后：
static CodResult cbpmGtsBroadcastTime(CodParamItem* item, CodText* value, CodParamScope scope)
{
    TRIM_WRAPPER(value);
    UPPER_VALUE(value);
    CodUint32 val;
    COD_CALL(anrParamSizeConvAndCheck(&item->name, value, &val,
                                      (CodUint64)ANH_MIN_GTS_BROADCAST_TIME, (CodUint64)ANH_MAX_GTS_BROADCAST_TIME));
    if (COD_PARAM_SCOPE_CONTAIN(scope, COD_PARAM_SCOPE_MEMORY)) {
        if (!item->isDeprecated) {
            AnlAttr* attr = anlGetAttr(gInstance->sql);
            attr->gtsBroadcastTime = val;
        } else if (!isParamSet(PARAM_GTS_BROADCAST_TIME) && !COD_PARAM_SCOPE_CONTAIN(scope, COD_PARAM_SCOPE_INIT)) {
            CodParamCtx*  ctx = anrGetParamCtx();
            CodParamItem* newItem = COD_DARRAY_AT(CodParamItem, &ctx->paramItems, PARAM_GTS_BROADCAST_TIME);

            if (codTextCopy(value, &newItem->value, newItem->bufSize) != COD_SUCCESS) {
                COD_SET_ERROR(ERR_CMM_PARAMETER_INVALID_VALUE, codErrT2S(&newItem->name), codErrT2SEx(&item->value));
                return COD_ERROR;
            }
    }
    if (item->isDeprecated) {
        logDeprecatedParam(item->name.str);
    }
    return COD_SUCCESS;
}
```

如果setSesParam不为空，也需要修改setSesParam。  若为过时参数，运行日志打印告警；若参数未过时，  更新SesParamCtx对应变量的值，  或者参数过时但未配置新参数同时在做修改而不是启动，则先将旧参数值转换给新参数后再  更新SesParamCtx对应变量的值。

  


**4.**  **loadParams阶段将paramId更改成新参数的id**  **，再赋值给attr对应变量，例：**

```
原： 
   COD_CALL(anrParamAsUint64(PARAM_DIN_RECONNECT_TIME, &attr->dinReconnectTime));
修改后：
```

**5.convertDeprecatedParams阶段对过时参数做兼容处理，对未配置新参数的过时参数，将参数值自动转换为新参数值，例：**

```
static CodResult processParamBroadcastTime(CodParamItem* item)
{
    if (isParamSet(PARAM_GTS_BROADCAST_TIME)) {
        return COD_SUCCESS;
    }

    COD_CALL(cbpmGtsBroadcastTime(newItem, &newItem->value, COD_PARAM_SCOPE_MEMORY));
    return COD_SUCCESS;
}
```
