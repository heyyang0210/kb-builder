####   [5.1.2 查询方式](#512-查询方式)  

所需信息：分布键类型/分布键

新增视图：DBA_DIST_TABLES/USER_DIST_TABLES/ALL_DIST_TABLES；DBA_DIST_KEY_COLUMNS/USER_DIST_KEY_COLUMNS/ALL_DIST_KEY_COLUMNS

```
--分布表
--dba视图
CREATE OR REPLACE VIEW DBA_DIST_TABLES(OWNER, TABLE_NAME, DIST_TYPE, DIST_KEY_COUNT) AS
    SELECT U.NAME, O.NAME, DECODE(D.TYPE#, 1, 'RANGE', 2, 'HASH', 3, 'LIST', 4, 'ROUNDROBIN',''), D.COLS 
    FROM SYS.OBJ$ O, SYS.USER$ U, SYS.TAB$ T, SYS.DIST$ D 
    WHERE T.OBJ# = D.OBJ# AND D.OBJ# = O.OBJ# AND O.OWNER# = U.USER#
/

CREATE OR REPLACE PUBLIC SYNONYM DBA_DIST_TABLES FOR SYS.DBA_DIST_TABLES
/
--user视图
CREATE OR REPLACE VIEW USER_DIST_TABLES(TABLE_NAME, DIST_TYPE, DIST_KEY_COUNT) AS
    SELECT O.NAME, DECODE(D.TYPE#, 1, 'RANGE', 2, 'HASH', 3, 'LIST', 4, 'ROUNDROBIN',''), D.COLS 
    FROM SYS.OBJ$ O, SYS.TAB$ T, SYS.DIST$ D 
    WHERE T.OBJ# = D.OBJ# AND D.OBJ# = O.OBJ# AND O.OWNER# = userenv('SCHEMAID')
/

CREATE OR REPLACE PUBLIC SYNONYM USER_DIST_TABLES FOR SYS.USER_DIST_TABLES
/
GRANT SELECT ON USER_DIST_TABLES TO PUBLIC
/

--all视图
CREATE OR REPLACE VIEW ALL_DIST_TABLES(OWNER, TABLE_NAME, DIST_TYPE, DIST_KEY_COUNT) AS
    SELECT U.NAME, O.NAME, DECODE(D.TYPE#, 1, 'RANGE', 2, 'HASH', 3, 'LIST', 4, 'ROUNDROBIN',''), D.COLS 
    FROM SYS.OBJ$ O, SYS.USER$ U, SYS.TAB$ T, SYS.DIST$ D 
    WHERE T.OBJ# = D.OBJ# AND D.OBJ# = O.OBJ# AND O.OWNER# = U.USER#
    AND (O.OWNER# = userenv('SCHEMAID')
         OR O.OBJ# IN (SELECT UOR.OBJ# FROM SYS.USER_OBJROLES$ UOR WHERE USER# = userenv('SCHEMAID'))
         OR check_sys_privilege(O.OWNER#, O.TYPE#) = 'TRUE')
/
CREATE OR REPLACE PUBLIC SYNONYM ALL_DIST_TABLES FOR SYS.ALL_DIST_TABLES
/
GRANT SELECT ON ALL_DIST_TABLES TO PUBLIC
/




--分布键
--dba视图
CREATE OR REPLACE VIEW DBA_DIST_KEY_COLUMNS(OWNER, NAME, COLUMN_NAME, COLUMN_POSITION) AS
SELECT U.NAME, O.NAME, C.NAME, PC.POS#
FROM SYS.USER$ U, SYS.OBJ$ O, SYS.COL$ C, SYS.DISTCOL$ PC
WHERE PC.OBJ#=O.OBJ# AND PC.OBJ#=C.OBJ# AND C.COL#=PC.COL# AND O.OWNER#= U.USER#
/
CREATE OR REPLACE PUBLIC SYNONYM DBA_DIST_KEY_COLUMNS FOR SYS.DBA_DIST_KEY_COLUMNS
/

--user视图
CREATE OR REPLACE VIEW USER_DIST_KEY_COLUMNS(NAME, COLUMN_NAME, COLUMN_POSITION) AS
SELECT O.NAME, C.NAME, PC.POS#
FROM SYS.OBJ$ O, SYS.COL$ C, SYS.DISTCOL$ PC
WHERE PC.OBJ#=O.OBJ# AND PC.OBJ#=C.OBJ# AND C.COL#=PC.COL# AND O.OWNER#= userenv('SCHEMAID')
/
CREATE OR REPLACE PUBLIC SYNONYM USER_DIST_KEY_COLUMNS FOR SYS.USER_DIST_KEY_COLUMNS
/
GRANT SELECT ON USER_DIST_KEY_COLUMNS TO PUBLIC
/


--all视图
CREATE OR REPLACE VIEW ALL_DIST_KEY_COLUMNS(OWNER, NAME, COLUMN_NAME, COLUMN_POSITION) AS
SELECT U.NAME, O.NAME, C.NAME, PC.POS#
FROM SYS.USER$ U, SYS.OBJ$ O, SYS.COL$ C, SYS.DISTCOL$ PC
WHERE PC.OBJ#=O.OBJ# AND PC.OBJ#=C.OBJ# AND C.COL#=PC.COL# AND O.OWNER#= U.USER#
AND (O.OWNER# = userenv('SCHEMAID')
     OR O.OBJ# IN (SELECT UOR.OBJ# FROM SYS.USER_OBJROLES$ UOR WHERE USER# = userenv('SCHEMAID'))
     OR check_sys_privilege(O.OWNER#, O.TYPE#) = 'TRUE')
/
CREATE OR REPLACE PUBLIC SYNONYM ALL_DIST_KEY_COLUMNS FOR SYS.ALL_DIST_KEY_COLUMNS
/
GRANT SELECT ON ALL_DIST_KEY_COLUMNS TO PUBLIC
/
```

--1.构造数据
--测试用例
drop user user1 cascade;
create user user1 identified by 1;
grant create session to user1;
grant create table to user1;
drop user user2 cascade;
create user user2 identified by 1;
grant create session to user2;
grant create table to user2;

--用户user1
conn user1/1@192.168.7.109:2688
--创建表,不带属性，默认为sharded
drop table if exists user1_test;
create table user1_test (id1 int, id2 char(10)) DISTRIBUTE BY HASH(id2);
insert into user1_test values(1,'aa');
insert into user1_test values(1,'bb');
--分布表+分区
drop table if exists user1_test1;
create sharded table user1_test1 (id1 int, id2 char(10)) DISTRIBUTE BY HASH(id2) partition by range(id1)(partition p1 values less than(1) ,partition p2 values less than(10));
insert into user1_test1 values(1,'aa');
insert into user1_test1 values(1,'bb');
--hash 分布键，多个
drop table if exists user1_test2;
create table user1_test2 (id1 int, id2 int, id3 char(10)) DISTRIBUTE BY HASH(id1,id2) partition by range(id2)(partition p1 values less than(1) ,partition p2 values less than(10));
insert into user1_test2 values(1,1,'aa');
insert into user1_test2 values(1,1,'bb');
--分区+lob
drop table if exists user1_test_lob1;
create table user1_test_lob1(a int, b clob,c clob,d int,e int) lob(C) store as (ENABLE STORAGE IN ROW) lob(B) store as (ENABLE STORAGE IN ROW) DISTRIBUTE BY HASH(a,d,e) partition by range(a)(partition p1 values less than(1) ,partition p2 values less than(10));
--复制表
drop table if exists user1_test_dup;
create duplicated table user1_test_dup (id1 int, id2 char(10))  organization tac partition by range(id1)(partition p1 values less than(1) ,partition p2 values less than(10));
insert into user1_test_dup values(1,'bb');
insert into user1_test_dup values(1,'aa');
commit;


--用户user2
conn user2/1@192.168.7.109:2688
--创建表
drop table if exists user2_test;
create table user2_test (id1 int, id2 char(10)) DISTRIBUTE BY HASH(id2);
insert into user2_test values(1,'aa');

--分布表+分区
drop table if exists user2_test1;

insert into user2_test1 values(1,'bb');
--hash 分布键，多个

insert into user2_test2 values(1,1,'bb');
--复制表
